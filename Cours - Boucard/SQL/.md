# 1. Lister toutes les plantes, avec affichage de tous les champs
`SELECT *
FROM PLANTE;`


# 2. Lister les plantes (n°, nom) dont le nom commence par R
`SELECT noplante, nomplante
FROM PLANTE
WHERE nomplante LIKE 'R%';`


# 3. Lister les plantes (noms), sans doublon
`SELECT distinct(nomplante)
FROM PLANTE;`


# 4. Lister les plantes (noms) de la région n°6
`SELECT nomplante
FROM PLANTE
WHERE noregion = 6;`


# 5. Lister les régions (noms) par ordre alphabétique descendant
`SELECT nomregion
FROM REGION
ORDER BY nomregion DESC;`


# 6. Afficher le nombre de serres dans le jardin
`SELECT COUNT(*) AS "Nombre de serres"
FROM SERRE;`


# 7. Afficher le nombre de bananiers dans le jardin
`SELECT COUNT(*) AS "Nombre de bananier"
FROM PLANTE
WHERE nomplante = 'Bananier';`


# 8. Afficher le nombre de plantes pour chaque région
`SELECT noregion, COUNT(*) AS "Nombre de plantes"
FROM PLANTE
GROUP BY noregion;`


# 9. Régions représentées par plus de 2 plantes
`SELECT noregion, COUNT(*) AS "Nombre de plantes"
FROM PLANTE
GROUP BY noregion
HAVING COUNT(*) > 2`


# 10. Plantes avec leur région d'origine
`SELECT P.nomplante, R.nomregion
FROM PLANTE P
JOIN REGION R ON P.noregion = R.noregion;`


# 11. Régions d'origine et plantes, triées par région puis plante
`SELECT R.nomregion, P.nomplante
FROM REGION R
JOIN PLANTE P ON R.noregion = P.noregion
ORDER BY R.nomregion ASC, P.nomplante ASC;`


# 12. Plantes de la serre n°3
`SELECT P.nomplante
FROM PLANTE P
JOIN EMPLACEMENT E ON P.noplante = E.noplante
WHERE E.noserre = 3
ORDER BY P.nomplante DESC;`


# 13. Plantes d'Afrique équatoriale et tropicale
`SELECT P.nomplante, R.nomregion
FROM PLANTE P
JOIN REGION R ON P.noregion = R.noregion
WHERE R.nomregion IN ('Afrique equatoriale', 'Afrique tropicale')
ORDER BY P.nomplante ASC;`


# 14. Plantes de la serre Exoticus (double jointure)
`SELECT P.nomplante
FROM PLANTE P
JOIN EMPLACEMENT E ON P.noplante = E.noplante
JOIN SERRE S ON E.noserre = S.noserre
WHERE S.nomserre = 'Exoticus';`


# 15. Nombre de plantes originaires d'Europe
`SELECT COUNT(*) AS "Nombre de plantes"
FROM PLANTE P
JOIN REGION R ON P.noregion = R.noregion
WHERE R.nomregion = 'Europe';`


# 16. Régions ayant au minimum 2 plantes
`SELECT R.noregion, R.nomregion, COUNT(P.noplante) AS "Nombre de plantes"
FROM REGION R
JOIN PLANTE P ON R.noregion = P.noregion
GROUP BY R.noregion, R.nomregion
HAVING COUNT(P.noplante) >= 2;`


# 17. Fréquence moyenne pour la serre Georges Bouvet
`SELECT AVG(V.frequence) AS "Fréquence Moyenne"
FROM VAPORISATION V
JOIN SERRE S ON V.noserre = S.noserre
WHERE S.nomserre = 'Georges Bouvet';`


# 18. Fréquence moyenne pour chaque serre
`SELECT S.noserre, S.nomserre,
       AVG(V.frequence) AS "Fréquence Moyenne"
FROM SERRE S
JOIN VAPORISATION V ON S.noserre = V.noserre
GROUP BY S.noserre, S.nomserre;`


# 19. Serres dont la fréquence moyenne est supérieure à 13
`SELECT S.nomserre AS "Nom Serre",
       AVG(V.frequence) AS "Fréquence Moyenne"
FROM SERRE S
JOIN VAPORISATION V ON S.noserre = V.noserre
GROUP BY S.noserre, S.nomserre
HAVING AVG(V.frequence) > 13;`


# 20. Plantes qui ne sont pas dans la serre Exoticus
`SELECT nomplante
FROM PLANTE
WHERE noplante NOT IN (
    SELECT E.noplante
    FROM EMPLACEMENT E
    JOIN SERRE S ON E.noserre = S.noserre
    WHERE S.nomserre = 'Exoticus'
);`


# 21. Nombre de plantes pour chaque région d'origine
`SELECT R.noregion, R.nomregion,
       COUNT(P.noplante) AS "Nombre de plantes"
FROM REGION R
LEFT JOIN PLANTE P ON R.noregion = P.noregion
GROUP BY R.noregion, R.nomregion;`


# 22. Nombre de plantes par serre
`SELECT S.noserre AS "n°Serre",
       S.nomserre AS "Nom Serre",
       COUNT(E.noplante) AS "Nombre"
FROM SERRE S
LEFT JOIN EMPLACEMENT E ON S.noserre = E.noserre
GROUP BY S.noserre, S.nomserre;`
